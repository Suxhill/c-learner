Program 05 - Concatenate The File

Problem Statement

Create 2 files and concatinate into 3rd file.

Source Code

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], f2[10], f3[10], data[50], a, *c;
	FILE * New, *New2, *New3;
	printf("Enter file name: ");
	gets(f);
	printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");
	New = fopen(f,"w");
	do{
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();}
        while(a != 19);
        printf("\nSaved successfully\n\n");
    fclose(New);

    printf("Enter 2nd file name: ");
	gets(f2);
	printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");
	New2 = fopen(f2,"w");
	do{
        gets(data);
        fprintf(New2, "%s\n", data);
        a = getch();}
        while(a != 19);
        printf("\nSaved successfully\n\n");
    fclose(New2);

    New = fopen(f, "r");
    New2 = fopen(f2, "r");
    printf("Enter 3rd file name: ");
    gets(f3);
    New3 = fopen(f3, "w");
    do{
    	c = fgets(data,50,New);
    	if(c != 0) fprintf(New3,"%s",data);
    }while(c != 0);
    do{
    	c = fgets(data,50,New2);
    	if(c != 0) fprintf(New3,"%s",data);
    }while(c != 0);
    fclose(New3);
    fclose(New2);
    fclose(New);
    printf("Your files are concatinated.");
}
